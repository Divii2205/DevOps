# Same alias, correct Host header -> real GitHub API response comes back:
PS D:\Users\user\Desktop\DevOps> kubectl exec -n s11-lab curl-client -- wget -qO- --no-check-certificate --header='Host: api.github.com' https://external-database-service/zen 2>&1 | head -3
